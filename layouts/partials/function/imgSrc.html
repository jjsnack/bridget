{{- /* Resolve an image resource to {url,w,h}, branching on environment. Prod (hugo.IsProduction + site.Params.cdnBase set): function/cdnSrc.html builds the CDN URL (default {cdnBase}/{key}{query}; a site may override it, and an empty result falls back to the preview below). Otherwise: Hugo-process the committed low-res copy via .Resize. Call with (dict "img" $img "page" $ownerPage "size" "hi"|"lo" ["spec" resizeSpec]). */ -}} {{- $out := dict -}} {{- if and hugo.IsProduction site.Params.cdnBase -}} {{- $out = partial "function/cdnSrc.html" . -}} {{- end -}} {{- if not $out.url -}} {{- $img := .img -}} {{- if ne site.Params.autoOrient false -}} {{- $img = $img | images.Filter (images.AutoOrient) -}} {{- end -}} {{- $spec := .spec -}} {{- if not $spec -}} {{- $spec = cond (eq .size "hi") (site.Params.hiResOpt | default "2500x webp Lanczos q75") (site.Params.loResOpt | default "800x webp Lanczos q60") -}} {{- end -}} {{- $r := $img.Resize $spec -}} {{- $out = dict "url" $r.RelPermalink "w" $r.Width "h" $r.Height -}} {{- end -}} {{- return $out -}}
