{{- /* CDN URL for an image resource, called by imgSrc.html in production when site.Params.cdnBase is set. Same args as imgSrc; returns (dict "url" U "w" W "h" H). Default: {cdnBase}/{key}{cdnHiQuery|cdnLoQuery}, object path = content-relative path or the resource's `cloudKey` param. A site overrides this partial for another URL scheme, or returns an empty dict to serve that image (or size) from the committed preview instead. */ -}} {{- $img := .img -}} {{- $obj := $img.Name -}} {{- with $img.Params.cloudKey -}} {{- $obj = . -}} {{- else -}} {{- if and .page .page.File -}} {{- $obj = path.Join .page.File.Dir $img.Name -}} {{- end -}} {{- end -}} {{- $q := cond (eq .size "hi") (site.Params.cdnHiQuery | default "") (site.Params.cdnLoQuery | default "") -}} {{- return (dict "url" (printf "%s/%s%s" site.Params.cdnBase $obj $q) "w" $img.Width "h" $img.Height) -}}
