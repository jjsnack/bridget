{{- /* Post image render hook. Markdown: ![caption](file.jpg "size") size ∈ small | medium (default) | large | wide | full — also used as flex weight inside {{< row >}}. Destination resolves as a page resource; otherwise the site may resolve it via function/imgSrcFallback.html (e.g. a key into an external image store); failing that, a raw src (static/remote). */ -}} {{- $size := .Title | default "medium" -}} {{- $alt := .Text -}} {{- $dest := .Destination -}} {{- $img := .Page.Resources.GetMatch $dest -}} {{- $page := .Page -}} {{- $lo := dict -}} {{- $hi := dict -}} {{- with $img -}} {{- $lo = partial "function/imgSrc.html" (dict "img" . "page" $page "size" "lo") -}} {{- $hi = partial "function/imgSrc.html" (dict "img" . "page" $page "size" "hi") -}} {{- else -}} {{- $lo = partial "function/imgSrcFallback.html" (dict "dest" $dest "page" $page "size" "lo") -}} {{- $hi = partial "function/imgSrcFallback.html" (dict "dest" $dest "page" $page "size" "hi") -}} {{- end -}}
